cmake_minimum_required(VERSION 3.22.1)
project(field_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/whisper ${CMAKE_BINARY_DIR}/whisper)
add_library(field_whisper SHARED whisper_jni.cpp)
target_link_libraries(field_whisper PRIVATE whisper android log)
target_link_options(field_whisper PRIVATE "-Wl,-z,max-page-size=16384")
